\noindent where the system we are interested in is the blue region, and the yellow and gray regions are dielectric materials with different permittivities.
The dielectric permittivity is then defined as a function of position:
\begin{equation}
    \epsilon(\mathbf{r}) = \begin{cases}
        \epsilon_1, & \mathbf{r} \in \Omega_1 \\
        \epsilon_2, & \mathbf{r} \in \Omega_2 \\
        \epsilon_0, & \mathbf{r} \in \Omega_0
    \end{cases}
\end{equation}

% We formulate the problem as calculating the four-index integral~\cite{cramer2013essentials} of the form:
Our goal is to compute the following four-index integral~\cite{cramer2013essentials}:
\begin{equation}
    \begin{split}
    V_{ijkl} &= \iint \varphi_i(\mathbf{r}_1) \varphi_j^*(\mathbf{r}_1) K(\mathbf{r}_1, \mathbf{r}_2) \varphi_k(\mathbf{r}_2) \varphi_l^*(\mathbf{r}_2) \mathrm{d}\mathbf{r}_1 \mathrm{d}\mathbf{r}_2\\
    % & = \int \rho_{ij}(\mathbf{r}_1) \left( \int K(\mathbf{r}_1, \mathbf{r}_2) \rho_{kl}(\mathbf{r}_2) \mathrm{d}\mathbf{r}_2 \right) \mathrm{d}\mathbf{r}_1, \\
    & = \int \rho_{ij}(\mathbf{r}_1) \phi_{kl}(\mathbf{r}_1) \mathrm{d}\mathbf{r}_1,
    \end{split}
\end{equation}
where~$\varphi_i(\mathbf{r})$ are the basis functions (orbitals), with a typical number of 20000,~$\rho_{ij}(\mathbf{r}) = \varphi_i(\mathbf{r}) \varphi_j^*(\mathbf{r})$ is the density distribution (may further be simplified as delta-like functions), and $\phi_{kl}(\mathbf{r})$ is the solution to the Poisson equation:
\begin{equation}\label{eq:poisson-origin}
    \grad_{\mathbf{r}} \cdot \left( \epsilon(\mathbf{r}) \grad_{\mathbf{r}} \phi_{kl}(\mathbf{r}) \right) = - \rho_{kl}(\mathbf{r}).
\end{equation}
We further assume that all density distributions are located inside the interested blue region, so that $\mathbf{r}_0 \in \Omega_0$.
Then Eq.~\eqref{eq:poisson-origin} can be rewritten as the following boundary value problem (indices omitted for simplicity):
\begin{equation}
    \begin{cases}
        \epsilon_0 \grad^2_{\mathbf{r}} \phi(\mathbf{r}) = - \rho(\mathbf{r})~, & \mathbf{r} \in \Omega_0 \\
        \epsilon_1 \grad^2_{\mathbf{r}} \phi(\mathbf{r}) = 0~, & \mathbf{r} \in \Omega_1 \\
        \epsilon_2 \grad^2_{\mathbf{r}} \phi(\mathbf{r}) = 0~, & \mathbf{r} \in \Omega_2 \\
        \phi(\mathbf{r})\mid_{\partial \Omega_i} = \phi(\mathbf{r})\mid_{\partial \Omega_j}~, & \mathbf{r} \in \partial \Omega_i \cap \partial \Omega_j \\
        \epsilon_i \mathbf{n} \cdot \grad_{\mathbf{r}} \phi(\mathbf{r})\mid_{\partial \Omega_i} = \epsilon_j \mathbf{n} \cdot \grad_{\mathbf{r}} \phi(\mathbf{r})\mid_{\partial \Omega_j}~, & \mathbf{r} \in \partial \Omega_i \cap \partial \Omega_j \\
        \phi(\mathbf{r}) \to 0~, & \abs{\mathbf{r}} \to + \infty \\
    \end{cases}
\end{equation}
where the first three equations are the Poisson equations in each region, the fourth is the continuity of potential at the interfaces between different regions, the fifth is the continuity of the normal component of the electric flux at the interfaces, and the last one is the infinite boundary condition.
Here, $\mathbf{n}$ denotes the unit outward normal on the interface.
By solving the above system, we can obtain the potential $\phi_{kl}(\mathbf{r})$, and then by integrating over the density distribution~$\rho_{ij}(\mathbf{r})$, we can obtain the four-index integral $V_{ijkl}$.

Specially, if the dielectric material in~$\Omega_1$ or~$\Omega_2$ can be treated as a perfect conductor, we have $\epsilon_m \to \infty$, and the equations becomes:
\begin{equation}\label{eq:poisson-perfect-conductor}
    \begin{cases}
        \epsilon_0 \grad^2_{\mathbf{r}} \phi(\mathbf{r}) = - \rho(\mathbf{r})~, & \mathbf{r} \in \Omega_0 \\
        \phi(\mathbf{r}) = 0~, & \mathbf{r} \in \Omega_1 \\
        \phi(\mathbf{r}) = 0~, & \mathbf{r} \in \Omega_2 \\
        % \mathbf{n} \times \grad_{\mathbf{r}} \phi(\mathbf{r})\mid_{\partial \Omega_m} = 0~, & \mathbf{r} \in \partial \Omega_m \\
        \phi(\mathbf{r})\mid_{\partial \Omega_i} = \phi(\mathbf{r})\mid_{\partial \Omega_0}~, & \mathbf{r} \in \partial \Omega_i \cap \partial \Omega_0 \\
        \epsilon_i \mathbf{n} \cdot \grad_{\mathbf{r}} \phi(\mathbf{r})\mid_{\partial \Omega_i} = \epsilon_0 \mathbf{n} \cdot \grad_{\mathbf{r}} \phi(\mathbf{r})\mid_{\partial \Omega_0}~, & \mathbf{r} \in \partial \Omega_i \cap \partial \Omega_0 \\
        \phi(\mathbf{r}) \to 0~, & \abs{\mathbf{r}} \to + \infty \\
    \end{cases}
\end{equation}
where the third and fourth equations enforce the perfect‑conductor conditions: the potential is zero inside the metal, and the field is normal to the surface.

\section{Solver}

In this section, we describe a boundary integral equation for the potential~$\phi(\mathbf{r})$ with a single layer potential~$\mathbf{\sigma}(\mathbf{r})$ on the boundaries.
The model is illustrated as follows:

\noindent where the gray region represents the density distribution~$\rho(\mathbf{r})$, and the arrows represent the normal direction of the boundaries.
The potential~$\phi(\mathbf{r})$ is separated into two parts:
\begin{equation}
    \phi(\mathbf{r}) = u(\mathbf{r}) + u_{\text{inc}}(\mathbf{r}),
\end{equation}
where~$u(\mathbf{r})$ is the scattered potential, and~$u_{\text{inc}}(\mathbf{r})$ is the incident potential, which are given by
\begin{equation}
    u_{\text{inc}}(\mathbf{r}) = \int \frac{\rho(\mathbf{r}')}{|\mathbf{r} - \mathbf{r}'|} \mathrm{d}\mathbf{r}'~,
    u(\mathbf{r}) = \int \frac{\mathbf{\sigma}(\mathbf{r}')}{|\mathbf{r} - \mathbf{r}'|} \mathrm{d}S(\mathbf{r}'),
\end{equation}
so that the continuity of the potential at the boundaries is automatically satisfied, and the continuity of the normal component of the electric flux requires that
\begin{equation}\label{eq:flux-continuity}
    \epsilon_i \partial_{\mathbf{n}} \left( u(\mathbf{r^+}) + u_{\text{inc}}(\mathbf{r^+}) \right) = \epsilon_j \partial_{\mathbf{n}} \left( u(\mathbf{r^-}) + u_{\text{inc}}(\mathbf{r^-}) \right),~\mathbf{r} \in \partial \Omega_i \cap \partial \Omega_j,
\end{equation}
where $\mathbf{r^+}$ and $\mathbf{r^-}$ denote the limit from two sides of the boundary.
Due to the existence of the single layer potential, the normal component of the scattered potential is given by
\begin{equation}\label{eq:scattered-potential}
    \begin{split}
    \partial_{\mathbf{n}} u(\mathbf{r^+}) &= \mathbf{n} \cdot \int \grad_{\mathbf{r}} \frac{\mathbf{\sigma}(\mathbf{r}')}{|\mathbf{r} - \mathbf{r}'|} \mathrm{d}S(\mathbf{r}') - \frac{\sigma(\mathbf{r})}{2}, = D \mathbf{\sigma}(\mathbf{r}) - \frac{\sigma(\mathbf{r})}{2} \\
    \partial_{\mathbf{n}} u(\mathbf{r^-}) &= \mathbf{n} \cdot \int \grad_{\mathbf{r}} \frac{\mathbf{\sigma}(\mathbf{r}')}{|\mathbf{r} - \mathbf{r}'|} \mathrm{d}S(\mathbf{r}') + \frac{\sigma(\mathbf{r})}{2} = D \mathbf{\sigma}(\mathbf{r}) + \frac{\sigma(\mathbf{r})}{2},
    \end{split}
\end{equation}
where $D$ represents the integral operator $\int \grad_{\mathbf{r}} \frac{\mathrm{d}S(\mathbf{r}')}{|\mathbf{r} - \mathbf{r}'|}$.
Then Eq.~\eqref{eq:flux-continuity} can be rewritten as
\begin{equation}
    \epsilon_i \left( D \mathbf{\sigma}(\mathbf{r}) - \frac{\sigma(\mathbf{r})}{2} + \partial_{\mathbf{n}} u_{\text{inc}}(\mathbf{r}) \right) = \epsilon_j \left( D \mathbf{\sigma}(\mathbf{r}) + \frac{\sigma(\mathbf{r})}{2} + \partial_{\mathbf{n}} u_{\text{inc}}(\mathbf{r}) \right),
\end{equation}
or equivalently,
\begin{equation}
    \frac{1}{2} \frac{\epsilon_i + \epsilon_j}{\epsilon_i - \epsilon_j} \sigma(\mathbf{r}) + D \sigma(\mathbf{r}) = - \partial_{\mathbf{n}} u_{\text{inc}}(\mathbf{r}),~\mathbf{r} \in \partial \Omega_i \cap \partial \Omega_j.
\end{equation}
By discretizing the three boundaries, the above equation can be written as a linear system of equations and solved iteratively.
