\noindent where the system we are interested in is the blue region, and the yellow and gray regions are dielectric materials with different permittivities.
The dielectric permittivity is then defined as a function of position:
\begin{equation}
    \epsilon(x) = \begin{cases}
        \epsilon_m, & x \in \Omega_m \\
        \epsilon_1, & x \in \Omega_1 \\
        \epsilon_2, & x \in \Omega_2 \\
        \epsilon_0 = 1, & x \in \Omega_0
    \end{cases}
\end{equation}

% We formulate the problem as calculating the four-index integral~\cite{cramer2013essentials} of the form:
Our goal is to compute the following four-index integral~\cite{cramer2013essentials}:
\begin{equation}
    \begin{split}
    V_{ijkl} &= \iint \varphi_i(x_1) \varphi_j^*(x_1) K(x_1, x_2) \varphi_k(x_2) \varphi_l^*(x_2) \mathrm{d}x_1 \mathrm{d}x_2\\
    & = \int \rho_{ij}(x_1) \phi_{kl}(x_1) \mathrm{d}x_1,
    \end{split}
\end{equation}
where~$\varphi_i(x)$ are the basis functions (orbitals), with a typical number of 20000,~$\rho_{ij}(x) = \varphi_i(x) \varphi_j^*(x)$ is the density distribution (may further be simplified as delta-like functions), and $\phi_{kl}(x)$ is the solution to the Poisson equation:
\begin{equation}\label{eq:poisson-origin}
    \grad_{x} \cdot \left( \epsilon(x) \grad_{x} \phi_{kl}(x) \right) = - \rho_{kl}(x).
\end{equation}
We further assume that all density distributions are located inside the interested blue region, so that $x_0 \in \Omega_I$.
Then Eq.~\eqref{eq:poisson-origin} can be rewritten as the following boundary value problem (indices omitted for simplicity):
\begin{equation}
    \begin{cases}
        \epsilon_I \Delta_{x} \phi(x) = - \rho(x)~, & x \in \Omega_m \\
        \Delta_{x} \phi(x) = 0~, & x \in \Omega_0 \cup \Omega_1 \cup \Omega_2 \\
        \phi(x)\mid_{\partial \Omega_i} = \phi(x)\mid_{\partial \Omega_j}~, & x \in \partial \Omega_i \cap \partial \Omega_j \\
        \epsilon_i \mathbf{n} \cdot \grad_{x} \phi(x)\mid_{\partial \Omega_i} = \epsilon_j \mathbf{n} \cdot \grad_{x} \phi(x)\mid_{\partial \Omega_j}~, & x \in \partial \Omega_i \cap \partial \Omega_j \\
        \phi(x) \to 0~, & \abs{x} \to + \infty \\
    \end{cases}
\end{equation}
where the first and second equations are for the region with or without source, the third and fifth are the boundary conditions at the interfaces between different regions, and the last one is the infinite boundary condition.
Here, $\mathbf{n}$ denotes the unit outward normal on the interface.
By solving the above system, we can obtain the potential $\phi_{kl}(x)$, and then by integrating over the density distribution~$\rho_{ij}(x)$, we can obtain the four-index integral $V_{ijkl}$.

\section{Solver}

In this section, we describe a boundary integral equation for the potential~$\phi(x)$ with a single layer potential~$\mathbf{\sigma}(x)$ on the boundaries.

\noindent where the gray region represents the density distribution~$\rho(x)$, and the arrows represent the normal direction of the boundaries (pointing from region with higher index to that with lower index).
The potential~$\phi(x)$ is separated into two parts:
\begin{equation}
    \phi(x) = u(x) + u_{\text{inc}}(x),
\end{equation}
where~$u(x)$ is the scattered potential, and~$u_{\text{inc}}(x)$ is the incident potential.
The incident potential satisfies:
\begin{equation}
    \begin{cases}
    \epsilon_m \Delta u_{\text{inc}}(x) = - \rho(x), & x \in \mathbb{R}^3 \\
    u_{\text{inc}}(x) \to 0, & \abs{x} \to + \infty 
    \end{cases}
\end{equation}
which is given by the following integral:
\begin{equation}
    u_{\text{inc}}(x) = \frac{1}{4 \pi \epsilon_m} \int \frac{\rho(x')}{|x - x'|} \mathrm{d}x'\;.
\end{equation}
Then the scattered potential satisfies:
\begin{equation}\label{eq:scattered-potential-boundary}
    \begin{cases}
    \Delta u(x) = 0, & x \in \Omega_i \\
    \epsilon_i \partial_{\mathbf{n}} \left( u^+(x) + u_{\text{inc}}(x) \right) = \epsilon_j \partial_{\mathbf{n}} \left( u^-(x) + u_{\text{inc}}(x) \right), & x \in \partial \Omega_i \cap \partial \Omega_j \\
    u(x) \to 0, & \abs{x} \to + \infty 
    \end{cases}
\end{equation}
which is a piecewise continuous function in each region.

Under the single layer potential representation, the scattered potential is given by
\begin{equation}
    u(x) = \frac{1}{4 \pi} \int_{\Gamma} \frac{\mathbf{\sigma}(x')}{|x - x'|} \mathrm{d}S(x'),
\end{equation}
where~$\Gamma$ is all boundaries in the system.
Due to the existence of the single layer potential, the normal component of the scattered potential is given by
\begin{equation}\label{eq:scattered-potential}
    \begin{split}
    \partial_{\mathbf{n}} u(\mathbf{r^+}) &= \frac{1}{4 \pi} \mathbf{n} \cdot \int \grad_{x} \frac{\mathbf{\sigma}(x')}{|x - x'|} \mathrm{d}S(x') - \frac{\sigma(x)}{2}, = D^T \mathbf{\sigma}(x) - \frac{\sigma(x)}{2} \\
    \partial_{\mathbf{n}} u(\mathbf{r^-}) &= \frac{1}{4 \pi} \mathbf{n} \cdot \int \grad_{x} \frac{\mathbf{\sigma}(x')}{|x - x'|} \mathrm{d}S(x') + \frac{\sigma(x)}{2} = D^T \mathbf{\sigma}(x) + \frac{\sigma(x)}{2},
    \end{split}
\end{equation}
where $D^T$ represents the integral operator $\int_{\Gamma} \grad_{x} \frac{\mathrm{d}S(x')}{|x - x'|}$.
Then the first and third equations of Eq.~\eqref{eq:scattered-potential-boundary} are automatically satisfied, and the second equation can be rewritten as
\begin{equation}
    \epsilon_i \left( D^T \mathbf{\sigma}(x) - \frac{\sigma(x)}{2} + \partial_{\mathbf{n}} u_{\text{inc}}(x) \right) = \epsilon_j \left( D^T \mathbf{\sigma}(x) + \frac{\sigma(x)}{2} + \partial_{\mathbf{n}} u_{\text{inc}}(x) \right),~x \in \partial \Omega_i \cap \partial \Omega_j.
\end{equation}
or equivalently,
\begin{equation}\label{eq:sl_equation}
    \frac{1}{2} \frac{\epsilon_i + \epsilon_j}{\epsilon_i - \epsilon_j} \sigma(x) + D^T \sigma(x) = - \partial_{\mathbf{n}} u_{\text{inc}}(x),~x \in \partial \Omega_i \cap \partial \Omega_j.
\end{equation}
The equation is a Fredholm integral equation of the second kind, and is identical to Eq.~(2.19) in Ref.~\cite{greengard1994numerical}.

Before discussing the discretization of the boundaries and the numerical results, we first consider the Gauss's law in 2D.

\begin{theorem}
    For systems shown in Fig.~\ref{fig:2d-box}, let~$\Omega_m \subset D \subset \Omega_0$, the following identity holds:
    \begin{equation}\label{eq:gauss-law-2d}
        \int_{\partial D} \partial_{\mathbf{n}} \phi(x) \mathrm{d}S(x) = q\;.
    \end{equation}
\end{theorem}

\begin{proof}
    Outside the box we have:
    \begin{equation}
        \int_{\partial D} \partial_{\mathbf{n}} \phi(x) \mathrm{d}S(x) = \int_{\partial \Omega_m} \partial_{\mathbf{n}} \phi^+(x) \mathrm{d}S(x)\;,
    \end{equation}
    inside the box we have:
    \begin{equation}
        \int_{\partial \Omega_m} \partial_{\mathbf{n}} \phi^-(x) \mathrm{d}S(x) = \frac{q}{\epsilon_m}\;.
    \end{equation}
    These identities are combined with the boundary condition that~$\epsilon_m \partial_{\mathbf{n}} \phi^-(x) = \partial_{\mathbf{n}} \phi^+(x)$ on the interface, we get Eq.~\eqref{eq:gauss-law-2d}.
\end{proof}

The single layer potential satisfies Eq.~\eqref{eq:sl_equation} on the boundaries, which needed to be discretized.
The simplest way is to discretize each edge into equal sized segments, and then use a Gauss-Legendre quadrature to evaluate the integral.
The results are shown in the right panel of Fig.~\ref{fig:2d-box}, where Eq.~\eqref{eq:gauss-law-2d} is used to validate the results.
I take~$\epsilon_m = 2.0$, fixed number of quadrature points to 16 in each panel and $d$ are the distances between the point charge and the boundaries.
A very bad convergence is observed especially when the point charge is close to the boundaries, as shown in Fig.~\ref{fig:2d-box-convergence-nonadaptive}.

To properly handle the corner singularities, we adaptively discretize the corners.
In each adaptation, we divide the panel near the corner into two panels, and then apply the Gauss-Legendre quadrature to the new panels.
The results are shown in Fig.~\ref{fig:2d-box-convergence-adaptive}, where the error decay exponentially to machine precision with the number of adaptations.

The contour plots of the potential is shown in Fig.~\ref{fig:2d-box-contourf}, for different values of~$\gamma = \frac{1 - \epsilon_m}{1 + \epsilon_m}$, where~$\epsilon_m \in (0, \infty)$.
A drop of accuracy is observed when $\gamma \to 1$, i.e., $\epsilon_m \to \infty$, corresponding to a perfect conductor.

% The value of the single layer density near the corner is shown in Fig.~\ref{fig:2d-box-density}, for values of~$\epsilon_m$ and distances~$d$.
In general, the single layer density near the corner is divergent in speed of
\begin{equation}
    \sigma(x) = O \left( x^{k} \right)\;,
\end{equation}
where~$k$ is dependent on the value of~$\gamma$.
The fitted blowup rate $k$ for different values of~$\gamma$ are shown in Fig.~\ref{fig:2d-box-blowup-analysis} (a), where the red and blue lines are predicted by the theory in Ref.~\cite{van2007electromagnetic}.
Specifically, when~$\gamma = \pm 1$, i.e. $\epsilon_m = \infty$ or $0$, ~$k = -1/3$.
A spectrum of the blowup rate is further shown in Fig.~\ref{fig:2d-box-blowup-analysis} (b), indicating that the density is contributed by a spectrum of modes, and the lowest mode is the divergent mode that dominates the density.

\section{Single Box in 3D space}

We then study a single cubic box with dielectric constant~$\epsilon_m$ in 3D space, assuming that a point charge~$q$ is located inside the box.
The equation is again given by Eq.~\ref{eq:sl_equation}.

The very first task is to do discretization for the surfaces of the cubic box, where we adaptively refine the edges and corners of the surfaces and apply tensor product Gauss-Legendre quadrature to the refined panels.
Two different strategies are considered: fixing the number of quadrature points on each panel or gradually reduce the number of points when the panel is close to the edges and corners.
The Resulting discretization is shown in Fig.~\ref{fig:3d-box-discretization}.

To validate the discretization, we make use of the Green's identity of the double layer operator:
\begin{equation}
    D 1 = - \frac{1}{2}\;,
\end{equation}
where $1$ represents a uniform density distribution.
We compared the results with different discretization strategies with the reference solution, as shown in Fig.~\ref{fig:3d-box-green-identity}.

We first checked the convergence of non-adaptive discretization with a large number of $p$.
Two tests are performed: the potential of the single layer density on a sphere with radius $2$ against a self converged reference solution and the total flux against~$1 - {\epsilon_m}^{-1}$.
In all tests, we set~$\epsilon_m = 4$ and the source charge is located at $(0.2, 0.3, 0.4)$.
The results are shown in Fig.~\ref{fig:3d-box-nonadapt-convergence}, indicating that order of the Gauss-Legendre quadrature has to be larger that~$16$ to start converging to the reference solution.

We then checked the convergence of adaptive discretization with fixed number of quadrature points, as shown in Fig.~\ref{fig:3d-box-adapt-convergence}.

However, although larger~$p$ can always gives a more accurate result when~$r$ is large enough.
